        \begin{column}{0.45\textwidth}
            \vspace{-0.5em}
            \begin{block}{Authentication Metrics}
                \fontsize{7pt}{8pt}\selectfont
                \begin{itemize}
                    \item \textbf{EER:} Point where False Acceptance and False Rejection rates are equal.
                    \item \textbf{FAR/FRR:} Standard frequency measures for unauthorized access and incorrect denials.
                \end{itemize}
            \end{block}

            \begin{block}{Privacy and Efficiency}
                \fontsize{7pt}{8pt}\selectfont
                \begin{itemize}
                    \item \textbf{Resistance to Attack:} Quantified by Reconstruction Error ($MSE_{recon}$).
                    \item \textbf{Revocability:} Validation of uncorrelated templates via key renewal.
                    \item \textbf{Latency ($L_{inf}$):} Total time from projection to decision ($<200$ms target).
                \end{itemize}
            \end{block}
